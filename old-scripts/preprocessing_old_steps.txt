    cmd = sprintf('%s/LN_GROW_LAYERS -rim %s/%s -N %d',LAYNII,MASKDIR,MASKS(i),NLAYERS);
    fprintf(fid,'%s\n\n',cmd);
end

cd(ANATDIR)
